//SELECTION Sort
suppose we have a array
{13,46,24,52,20,9}
{9,46,24,52,20,13}STEP 1
{9,13,24,52,20,46}STEP 2
{9,13,20,52,24,46}STEP 3
{9,13,20,24,52,46}STEP 4
{9,13,20,24,46,52}STEP 5---. ARRAY IS SORTED AT STEP 5

#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin>>n;
    int arr[n];
    for(int i=0;i<n;i++) {
        cin>>arr[i];
    }
    for(int i=0;i<n-2;i++) {
        int min=arr[i];
        for(int j=0;j<n-1;j++) {
            if(arr[i]<arr[j] ) {
                swap(arr[i],arr[j]);
            }
        }
    }
    for(int i=0;i<n:i++) {
        cout<<arr[i];
    }
    return 0;
} //TIME COMPLEXITY O(N^2);



//BUBBLE SORT
{PUSHES THE MAXIMUM TO THE LAST}

#include<bits/stdc++.h>
using namespace std;

int main() {
    int n;
    cin>>n;
    int arr[n];
    for(int i=0;i<n;i++) {
        cin>>arr[i];
    }
    for(int i=(n-1);i>0;i=i-1) {
        int didswap=0;
        for(int j=0;j<=i;j=j+1) {
           
            if(arr[j]>arr[j+1]) {

            swap(arr[j],arr[j+1]);
          didswap=1;
        }
        if(didswap==0) {
            break;
        }
        }
    }
    for(int i=0;i<n;i++) {
        cout<<arr[i]<<" ";
    }
    return 0;
}
